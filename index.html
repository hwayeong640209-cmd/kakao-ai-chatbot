import fs from "fs"; import path from "path"; import { GoogleGenAI } from "@google/genai"; const ai = new GoogleGenAI({ apiKey: process.env.GEMINI_API_KEY, }); // 챗봇22.txt 읽기 function loadKnowledge() { const filePath = path.join(process.cwd(), "data", "챗봇22.txt"); if (!fs.existsSync(filePath)) { throw new Error("data/챗봇22.txt 파일을 찾을 수 없습니다."); } return fs.readFileSync(filePath, "utf8"); } // 검색용 단어 정리 function normalize(text) { return text .toLowerCase() .replace(/[?!.,。！？]/g, " ") .replace(/\s+/g, " ") .trim(); } // 불필요한 조사 제거 function tokenize(text) { const stopWords = [ "은", "는", "이", "가", "을", "를", "에", "에서", "으로", "로", "와", "과", "도", "의", "하고", "좀", "요", "해주세요", "알려줘", "알려주세요", "어떻게", "뭐", "무엇", ]; return normalize(text) .split(" ") .filter(Boolean) .map((word) => { let result = word; for (const stopWord of stopWords) { if (result.endsWith(stopWord) && result.length > stopWord.length) { result = result.slice(0, -stopWord.length); } } return result; }) .filter((word) => word.length >= 1); } // 자료를 문단 단위로 나누기 function splitKnowledge(text) { return text .split(/\n\s*\n/) .map((section) => section.trim()) .filter((section) => section.length > 0); } // 질문과 자료의 관련도 계산 function scoreSection(question, section) { const questionWords = tokenize(question); const normalizedSection = normalize(section); let score = 0; for (const word of questionWords) { if (word.length < 1) continue; if (normalizedSection.includes(word)) { score += word.length >= 2 ? 3 : 1; } } return score; } // 질문과 관련된 자료만 추출 function findRelevantKnowledge(question, knowledge) { const sections = splitKnowledge(knowledge); const scored = sections .map((section) => ({ section, score: scoreSection(question, section), })) .sort((a, b) => b.score - a.score); // 아무것도 검색되지 않은 경우 if (!scored.length || scored[0].score === 0) { return ""; } // 관련도가 높은 자료만 전달 const selected = scored .filter((item) => item.score > 0) .slice(0, 8) .map((item) => item.section); return selected.join("\n\n--------------------\n\n"); } export default async function handler(req, res) { if (req.method !== "POST") { return res.status(405).json({ error: "POST 요청만 허용됩니다.", }); } try { const { message } = req.body || {}; if (!message || typeof message !== "string") { return res.status(400).json({ error: "메시지를 입력해주세요.", }); } const knowledge = loadKnowledge(); // 인사말은 별도로 처리 const greetingPattern = /^(안녕|안녕하세요|하이|hello|hi|반가워|반갑습니다)[!?.\s]*$/i; if (greetingPattern.test(message.trim())) { return res.status(200).json({ answer: "안녕하세요! 한국도로교통공단 전남운전면허시험장 업무 안내 챗봇입니다. 운전면허와 관련하여 궁금하신 업무를 말씀해 주시면 안내해 드리겠습니다.", }); } // 질문과 관련된 업무자료 검색 const relevantKnowledge = findRelevantKnowledge( message, knowledge ); // 관련 자료를 찾지 못한 경우 if (!relevantKnowledge) { return res.status(200).json({ answer: "죄송합니다. 제공된 업무자료에서 해당 질문과 관련된 내용을 찾지 못했습니다. 전남운전면허시험장 업무와 관련된 질문을 해주세요.", }); } const systemInstruction = ` 당신은 한국도로교통공단 전남운전면허시험장의 업무 안내 챗봇입니다. 가장 중요한 원칙은 다음과 같습니다. 1. 반드시 아래에 제공된 [업무자료]만 근거로 답변하세요. 2. 업무자료에 없는 사실을 절대로 추가하지 마세요. 일반적인 상식이나 인터넷에서 알고 있는 내용을 사용하지 마세요. 3. 업무자료에 없는 질문이라면 "제공된 업무자료에서 해당 내용을 확인할 수 없습니다." 라고 답변하세요. 4. 업무자료의 숫자, 시간, 수수료, 면제 여부, 준비물, 절차 등을 임의로 변경하지 마세요. 5. 업무자료에 서로 다른 내용이 함께 존재하는 경우, 임의로 판단하거나 하나로 합치지 마세요. 질문과 가장 직접적으로 관련된 항목을 기준으로 답변하고, 필요하다면 "업무자료에 관련 내용이 다르게 기재되어 있어 정확한 확인이 필요합니다."라고 안내하세요. 6. 사용자가 질문한 내용에 필요한 정보만 간결하게 답변하세요. 7. 답변은 한국어로 하세요. 8. "[핵심]", "[주의사항]" 같은 내부 태그를 사용하지 마세요. 9. 제공된 자료에 URL이 있다면 해당 URL을 그대로 안내할 수 있습니다. 단, 새로운 URL을 임의로 만들어내지 마세요. 10. 사용자의 질문에 직접 답하고, 필요할 경우 준비물 / 수수료 / 절차 등을 bullet 형식으로 정리하세요. [업무자료] ${relevantKnowledge} `; const response = await ai.models.generateContent({ model: "gemini-3.5-flash-lite", contents: message, config: { systemInstruction, temperature: 0.1, }, }); const answer = response.text; return res.status(200).json({ answer: answer || "죄송합니다. 제공된 업무자료를 바탕으로 답변을 생성하지 못했습니다.", }); } catch (error) { console.error("Gemini API Error:", error); return res.status(500).json({ error: error?.message || "Gemini API 호출 중 오류가 발생했습니다.", }); } }
